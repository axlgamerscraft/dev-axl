local HTTP = (http_request or request or (syn and syn.request) or (http and http.request))

local function httpRequest(opts)
	if HTTP then
		local ok, res = pcall(HTTP, opts)
		if ok then return res end
	else
		if opts.Method == "POST" or opts.Method == "DELETE" then
			local ok, res = pcall(HttpService.PostAsync, HttpService, opts.Url, opts.Body or "")
			if ok then return {Success=true, Body=res} end
		else
			local ok, res = pcall(HttpService.GetAsync, HttpService, opts.Url)
			if ok then return {Success=true, Body=res} end
		end
	end
	return nil
end

local function RealTranslate(text, target)
	if target == "en" then return text end
	local key = target .. "|" .. text
	if TransCache[key] then return TransCache[key] end
	local url = "https://translate.googleapis.com/translate_a/single"
		.. "?client=gtx&sl=auto&tl=" .. target .. "&dt=t&q=" .. HttpService:UrlEncode(text)
	local res = httpRequest({Url = url, Method = "GET"})
	if res and res.Success and res.Body then
		local ok, data = pcall(HttpService.JSONDecode, HttpService, res.Body)
		if ok and type(data) == "table" and data[1] then
			local out = {}
			for _, seg in ipairs(data[1]) do
				if type(seg) == "table" and seg[1] then table.insert(out, tostring(seg[1])) end
			end
			local result = table.concat(out, "")
			if result ~= "" then TransCache[key] = result return result end
		end
	end
	return text
end

local function getModule2(name)
	for _, d in ipairs(RepStorage:GetDescendants()) do
		if d:IsA("ModuleScript") and d.Name == name then
			local ok, mod = pcall(require, d)
			if ok then return mod end
		end
	end
	return nil
end

local AssetsData = getModule2("Assets")
local EggStateModule = getModule2("EggState")

local function resolveRarity(category)
	if AssetsData and AssetsData.Directory and AssetsData.Directory[category] then
		local entry = AssetsData.Directory[category]
		local r = entry and entry.Rarity
		if type(r) == "table" then return tostring(r._id or r.DisplayName) end
		if type(r) == "string" then return r end
	end
	return "Common"
end

local function getAssetName(category)
	if AssetsData and AssetsData.Directory and AssetsData.Directory[category] then
		local entry = AssetsData.Directory[category]
		if entry and entry.DisplayName then return tostring(entry.DisplayName) end
	end
	return tostring(category or "Unknown")
end

local function collectLocalEggs()
	local list = {}
	if EggStateModule and type(EggStateModule.ReadFieldEggs) == "function" then
		local ok, snap = pcall(EggStateModule.ReadFieldEggs)
		if ok and snap and snap.Records then
			for _, rec in pairs(snap.Records) do
				local cat = rec.AssetCategory or "Unknown"
				table.insert(list, {
					name = getAssetName(cat),
					rarity = resolveRarity(cat),
					area = tostring(rec.AreaId or "Map")
				})
			end
		end
	end
	return list
end

local function bestRarity(eggs)
	local best, bestRank = "Common", 0
	for _, e in ipairs(eggs) do
		local rk = RARITY_ORDER[e.rarity] or 1
		if rk > bestRank then best, bestRank = e.rarity, rk end
	end
	return best
end

local function isoToStamp(iso)
	-- يحول ISO time إلى unix seconds
	local y, mo, d, h, mi, s = iso:match("(%d+)-(%d+)-(%d+)T(%d+):(%d+):(%d+)")
	if not y then return 0 end
	return os.time({year=tonumber(y), month=tonumber(mo), day=tonumber(d),
		hour=tonumber(h), min=tonumber(mi), sec=tonumber(s)})
end

local function isStale(server)
	local iso = server.updated_at
	if not iso or iso == "" then return true end
	local age = os.time(os.date("!*t")) - isoToStamp(iso)
	return age > SERVER_TTL
end

local function pushToSupabase(eggs, playerCount)

		best_rarity = bestRarity(eggs),

	})
	httpRequest({
		Url = SUPABASE_URL .. "/rest/v1/servers?on_conflict=jobid",

			["Content-Type"]="application/json",

			["Authorization"]="Bearer "..SUPABASE_KEY,
			["Prefer"]="resolution=merge-duplicates"

	})
end

local function deleteStaleServersFromSupabase()
	if not HttpEnabled then return end
	-- نقطة الوقت: الآن ناقص 60 ثانية
	local cutoff = os.date("!%Y-%m-%dT%H:%M:%SZ", os.time(os.date("!*t")) - SERVER_TTL)
	local url = SUPABASE_URL
		.. "/rest/v1/servers?updated_at=lt." .. cutoff
	httpRequest({
		Url = url,
		Method = "DELETE",
		Headers = {
			["apikey"]=SUPABASE_KEY,
			["Authorization"]="Bearer "..SUPABASE_KEY,
			["Prefer"]="return=minimal"
		}
	})
end

local function fetchServers()
	if not HttpEnabled then return {} end
	local res = httpRequest({
		Url = SUPABASE_URL .. "/rest/v1/servers?order=updated_at.desc&limit=" .. MAX_SERVERS,
		Method = "GET",
		Headers = {["apikey"]=SUPABASE_KEY, ["Authorization"]="Bearer "..SUPABASE_KEY}
	})
	if res and res.Success and res.Body then
		local ok, data = pcall(HttpService.JSONDecode, HttpService, res.Body)
		if ok and type(data) == "table" then
			-- فلترة محلية أيضاً: استبعد الأقدم من دقيقة
			local fresh = {}
			for _, s in ipairs(data) do
				if not isStale(s) then table.insert(fresh, s) end
			end
			return fresh
		end
	end
	return {}
end

end

local BlurEnabled = true
local blurTick = 0

local function UpdateBlur(newPos)
	if not BlurEnabled then return end
	blurTick = blurTick + 1
	-- كل إطار نحرك الأشباح بالتتابع (تتبع متأخر)
	for i = 1, BLUR_GHOST_COUNT do
		local g = BlurGhosts[i]
		if i == 1 then
			g.pos = Main.Position
		else
			local prev = BlurGhosts[i-1]
			g.pos = g.pos:Lerp(prev.pos, 0.35)
		end
		g.frame.Position = g.pos
		local depth = i / BLUR_GHOST_COUNT
		g.frame.BackgroundTransparency = 0.94 - (1 - depth) * 0.06
		g.frame.Size = Main.Size
	end
end

local function ResetBlur()
	for _, g in ipairs(BlurGhosts) do
		g.frame.BackgroundTransparency = 1
		g.pos = Main.Position
	end
end

local function parseEggs(raw)

	local ok, data = pcall(HttpService.JSONDecode, HttpService, tostring(raw or "[]"))
	return ok and data or {}
end

local function BuildServerCard(index, server)
	local eggs = parseEggs(server.eggs)
	local br = server.best_rarity or bestRarity(eggs)
	local color = RARITY_COLORS[br] or RARITY_COLORS.Common

	local card = Instance.new("Frame")
	card.Size = UDim2.new(1,-8,0,120)

	card.BorderSizePixel = 0; card.Parent = ServerScroll

	local stroke = Instance.new("UIStroke")
	stroke.Color = color; stroke.Transparency = .55; stroke.Thickness = 1.2; stroke.Parent = card

	local num = Instance.new("TextLabel")
	num.Size = UDim2.new(1,-130,0,20); num.Position = UDim2.fromOffset(14,10)
	num.BackgroundTransparency = 1
	num.Text = string.format("%s #%d • %s", L("SERVER"), index, br)
	num.TextColor3 = color; num.TextSize = 13; num.Font = Enum.Font.GothamBold
	num.TextXAlignment = Enum.TextXAlignment.Left; num.Parent = card

	-- عمر السيرفر بالثواني
	local ageTag = Instance.new("TextLabel")
	ageTag.Size = UDim2.fromOffset(60,20); ageTag.Position = UDim2.new(1,-240,0,10)

	playersTag.Text = "👤 " .. tostring(server.players or "?") .. "/" .. tostring(server.max_players or "?")

	playersTag.Font = Enum.Font.GothamBold; playersTag.TextXAlignment = Enum.TextXAlignment.Right

		Pop()

		local pid = tonumber(server.place_id) or game.PlaceId
		if jid == "" or jid:find("_local") then
			joinBtn.Text = "✗ INVALID"
			task.wait(1.2); joinBtn.Text = "→ " .. L("JOIN")
			return
		end

			queue_on_teleport(SCRIPT_SOURCE)
		end

			TeleportService:TeleportToPlaceInstance(pid, jid, Player)
		end)
		if not ok then
			joinBtn.Text = "✗ FAILED"
			task.wait(1.5); joinBtn.Text = "→ " .. L("JOIN")
		end

		while card.Parent do
			local p = (math.sin(os.clock()*2) + 1) / 2
			stroke.Transparency = .55 - p * .25
			task.wait(.05)
		end
	end)

local function RefreshServers()
	StatusLabel.Text = L("WAITING")
	for _, c in ipairs(ServerCards) do c:Destroy() end
	table.clear(ServerCards)

	task.spawn(function()
		local servers = fetchServers()
		local idx = 0
		for _, s in ipairs(servers) do
			if idx >= MAX_SERVERS then break end
			local br = s.best_rarity or "Common"
			if CurrentFilterRank == 0 or (RARITY_ORDER[br] or 0) >= CurrentFilterRank then
				idx = idx + 1
				BuildServerCard(idx, s)
			end
		end
		StatusLabel.Text = idx > 0 and (L("FOUND") .. ": " .. idx) or L("NO_SERVERS")
	end)
end

-- العداد التنازلي للتنظيف + تحديث تلقائي كل 60 ثانية حقيقية
task.spawn(function()
	local remaining = CLEAN_INTERVAL
	while true do
		task.wait(1) -- ثانية حقيقية
		if not EggPage.Visible then continue end
		remaining = remaining - 1
		if remaining <= 0 then
			remaining = CLEAN_INTERVAL
			-- 1) حذف القديمة من Supabase
			if HttpEnabled then pcall(deleteStaleServersFromSupabase) end
			-- 2) إعادة بناء الصفحة (تُشيل الكروت القديمة تلقائياً)
			if EggPage.Visible then
				RefreshServers()
			end
		end
		-- تحديث العداد على الشاشة
		CleanCountdown.Text = "🧹 " .. remaining .. "s"
		if remaining <= 5 then
			CleanCountdown.TextColor3 = Color3.fromRGB(255,120,120)
		else
			CleanCountdown.TextColor3 = Color3.fromRGB(130,200,255)
		end
	end
end)

	RefreshServers()
end)

local function MakeDraggable(object)
	local dragging, dragStart, startPosition = false, nil, nil
	object.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true; dragStart = input.Position; startPosition = object.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
					-- عند الإفلات: أطفئ الأشباح بنعومة
					for _, g in ipairs(BlurGhosts) do
						Tween(g.frame, .25, {BackgroundTransparency = 1})
					end
				end
			end)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if not dragging then return end
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			local delta = input.Position - dragStart
			object.Position = UDim2.new(startPosition.X.Scale, startPosition.X.Offset + delta.X,
				startPosition.Y.Scale, startPosition.Y.Offset + delta.Y)
			-- تحريك أثر الـ Motion Blur
			UpdateBlur(object.Position)
		end
	end)
end
